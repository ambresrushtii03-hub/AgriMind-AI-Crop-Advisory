DATABASE SCHEMA (PRODUCTION SQL FOR POSTGRESQL)Execute this exact schema on Replit Postgres / PostgreSQL. Ensure table relationships, constraints, foreign key cascades, indices, and timestamps are strictly configured.SQL-- Enable UUID extension if not already present
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 1. USERS TABLE
CREATE TABLE IF NOT EXISTS users (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    email VARCHAR(255) UNIQUE NOT NULL,
    full_name VARCHAR(255) NOT NULL,
    preferred_language VARCHAR(10) DEFAULT 'en',
    measurement_system VARCHAR(10) DEFAULT 'metric', -- 'metric' or 'imperial'
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 2. FARMS TABLE
CREATE TABLE IF NOT EXISTS farms (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    name VARCHAR(255) NOT NULL,
    location_name VARCHAR(255) NOT NULL,
    latitude DECIMAL(10, 8),
    longitude DECIMAL(11, 8),
    total_area_hectares DECIMAL(10, 2) NOT NULL,
    default_soil_type VARCHAR(50) DEFAULT 'Loam',
    irrigation_type VARCHAR(50) DEFAULT 'Drip', -- 'Drip', 'Sprinkler', 'Rainfed', 'Flood'
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 3. FIELD ZONES TABLE
CREATE TABLE IF NOT EXISTS field_zones (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    farm_id UUID NOT NULL REFERENCES farms(id) ON DELETE CASCADE,
    zone_name VARCHAR(255) NOT NULL,
    area_hectares DECIMAL(10, 2) NOT NULL,
    current_crop VARCHAR(255) NOT NULL,
    crop_cultivar VARCHAR(255),
    planting_date DATE,
    expected_harvest_date DATE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 4. SOIL TESTS TABLE
CREATE TABLE IF NOT EXISTS soil_tests (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    zone_id UUID NOT NULL REFERENCES field_zones(id) ON DELETE CASCADE,
    ph_level DECIMAL(4, 2) NOT NULL,
    nitrogen_ppm DECIMAL(8, 2) NOT NULL,
    phosphorus_ppm DECIMAL(8, 2) NOT NULL,
    potassium_ppm DECIMAL(8, 2) NOT NULL,
    organic_matter_percent DECIMAL(4, 2),
    electrical_conductivity_ds_m DECIMAL(6, 2),
    tested_at DATE NOT NULL DEFAULT CURRENT_DATE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 5. ADVISORIES TABLE
CREATE TABLE IF NOT EXISTS advisories (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    farm_id UUID REFERENCES farms(id) ON DELETE SET NULL,
    zone_id UUID REFERENCES field_zones(id) ON DELETE SET NULL,
    crop_name VARCHAR(255) NOT NULL,
    growth_stage VARCHAR(100) NOT NULL,
    urgency_level VARCHAR(20) NOT NULL CHECK (urgency_level IN ('LOW', 'MEDIUM', 'HIGH', 'CRITICAL')),
    primary_diagnosis VARCHAR(255) NOT NULL,
    confidence_score DECIMAL(5, 2) NOT NULL, -- e.g., 94.50
    summary TEXT NOT NULL,
    full_advisory_json JSONB NOT NULL,
    symptoms_observed TEXT[],
    image_url TEXT,
    language_code VARCHAR(10) DEFAULT 'en',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 6. CHAT SESSIONS TABLE
CREATE TABLE IF NOT EXISTS chat_sessions (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    farm_id UUID REFERENCES farms(id) ON DELETE SET NULL,
    title VARCHAR(255) NOT NULL DEFAULT 'Agronomic Consultation',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 7. CHAT MESSAGES TABLE
CREATE TABLE IF NOT EXISTS chat_messages (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    session_id UUID NOT NULL REFERENCES chat_sessions(id) ON DELETE CASCADE,
    sender VARCHAR(20) NOT NULL CHECK (sender IN ('user', 'assistant', 'system')),
    content TEXT NOT NULL,
    metadata_json JSONB DEFAULT '{}'::jsonb,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- INDEXES FOR HIGH-PERFORMANCE QUERYING
CREATE INDEX IF NOT EXISTS idx_farms_user_id ON farms(user_id);
CREATE INDEX IF NOT EXISTS idx_field_zones_farm_id ON field_zones(farm_id);
CREATE INDEX IF NOT EXISTS idx_soil_tests_zone_id ON soil_tests(zone_id);
CREATE INDEX IF NOT EXISTS idx_advisories_user_id ON advisories(user_id);
CREATE INDEX IF NOT EXISTS idx_advisories_farm_id ON advisories(farm_id);
CREATE INDEX IF NOT EXISTS idx_advisories_urgency ON advisories(urgency_level);
CREATE INDEX IF NOT EXISTS idx_chat_messages_session_id ON chat_messages(session_id);
